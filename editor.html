function copyCode() {
  const html = document.getElementById('htmlPre').textContent;
  const css = document.getElementById('cssPre').textContent;
  const js = document.getElementById('jsPre').textContent;
  const fullCode = `\n${html}\n\n\n${css}\n\n\n${js}`;
  navigator.clipboard.writeText(fullCode);
  updateStatus('Code copied to clipboard!');
}
